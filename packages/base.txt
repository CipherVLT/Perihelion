# Perihelion base: every flavor gets these, and nothing else is added
# behind your back: %packages runs with --exclude-weakdeps, so packages
# Fedora only "recommends" are not installed. If something seems missing
# after an install, it was probably a weak dependency: add it here.
#
# One package or @group per line. Anything missing fails the install.
# Run scripts/check-packages.sh on a Fedora machine to check a list early.

# --- System ------------------------------------------------------------------
# Fedora's smallest bootable system: kernel, systemd, dnf, sudo, basic tools
@core
# Weak dependencies of @core that a desktop can't do without (systemd and
# glibc only "recommend" them, so --exclude-weakdeps drops them):
#   systemd-pam: pam_systemd.so registers logins with logind. Without it
#     there is no seat for the login screen or any Wayland session.
#   glibc-langpack-en: an English UTF-8 locale (otherwise everything runs in "C")
systemd-pam
glibc-langpack-en
#   dconf: where GTK apps keep their settings (dark mode, Strata's theme sync)
#   cracklib-dicts: passwd can't check (or set) new passwords without it
#   glibc-gconv-extra: character-set converters some apps and games need
dconf
cracklib-dicts
glibc-gconv-extra
# A terminal text editor (sudoedit needs one: nano-default-editor)
nano
nano-default-editor
# System info in every new terminal (Veyl OS layout; perihelion-fastfetch.sh)
# Keeps the clock right over the internet (NTP); without it the clock is
# only as good as the hardware clock
chrony
fastfetch
# Terminal audio visualizer (Catppuccin gradient config seeded)
cava
# Hardware sensors (sensors, sensors-detect) and per-app network usage, as
# system monitors like Mission Center expect
lm_sensors
nethogs
# CPU microcode and firmware for Wi-Fi, Bluetooth and GPUs (laptops need it)
@hardware-support
# `dnf copr` (the Hyprland flavor keeps its COPR enabled with it)
dnf5-plugins

# --- Network, Bluetooth, firewall --------------------------------------------
NetworkManager
NetworkManager-wifi
NetworkManager-bluetooth
# Wi-Fi from a terminal (nmtui); Event Horizon's network card opens it too
NetworkManager-tui
firewalld
bluez

# --- Audio -------------------------------------------------------------------
pipewire
pipewire-alsa
pipewire-pulseaudio
wireplumber

# --- Graphics ----------------------------------------------------------------
# 64-bit drivers, arch spelled out: gaming.txt also asks for the .i686
# builds, and with both listed dnf can resolve the bare name to i686 only
# (that left the system without a 64-bit Vulkan driver).
mesa-dri-drivers.x86_64
mesa-vulkan-drivers.x86_64
vulkan-loader.x86_64
# X11 apps (Steam, most games)
xorg-x11-server-Xwayland

# --- Login screen: SDDM on a Wayland greeter + Pixie theme ---------------------
sddm
# The greeter runs on Weston (sddm-wayland-generic) and is a Qt 6 Wayland
# client; with weak dependencies off, list both explicitly.
sddm-wayland-generic
weston
qt6-qtwayland
qt6-qtdeclarative
qt6-qtsvg

# --- Desktop plumbing (shared by all compositors) -----------------------------
xdg-desktop-portal
xdg-desktop-portal-gtk
xdg-user-dirs
xdg-utils
polkit
gnome-keyring
gnome-keyring-pam
# Battery and power profiles (laptops)
power-profiles-daemon
upower
# Laptop screen brightness keys
brightnessctl
desktop-file-utils

# Flatpak, with the Flathub remote added for you (build/common-post.sh and
# perihelion-flathub.service), so `flatpak install flathub <app>` just works
flatpak

# --- Apps ----------------------------------------------------------------------
kitty
firefox
fuzzel
# App drawer grid (perihelion-drawer, Super+A)
# nwg-drawer itself is built by scripts/build-nwg-drawer.sh (Fedora dropped it);
# these are the libraries it links against.
gtk3
gtk-layer-shell
# Media keys (play/pause, next, previous) in every flavor
playerctl
# Strata (the file manager) comes from its release via build/install-strata.sh;
# these are the libraries it links against. gvfs provides the Trash (trash:///).
gtk4
gtksourceview5
poppler-glib
graphene
gstreamer1
gstreamer1-plugins-base
gvfs
bubblewrap

# --- Fonts -----------------------------------------------------------------------
adwaita-icon-theme
# Look: Bibata cursor (Terra), DejaVu Sans Mono (terminal). Numix-Circle

bibata-cursor-theme
dejavu-sans-mono-fonts

qt6ct

imv
dejavu-lgc-sans-fonts
hicolor-icon-theme
google-noto-sans-fonts
google-noto-sans-mono-fonts
google-noto-color-emoji-fonts

terra-release
